{% trans "Start the path with /fr/, /es/, /pt/ or /de/ to open the page in that language, as in /fr/import/build/; without it, the page follows the reader's browser. A build for another game than the address opens on its own game." %}

const bytes = new TextEncoder().encode(JSON.stringify(build));
let binary = '';
bytes.forEach(function (byte) { binary += String.fromCharCode(byte); });
const data = btoa(binary).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
const href = '{{ site_url }}/import/build/?data=' + data;
