# stays 3.9, so install with python3.14, the version the Docker image runs.
# mod-wsgi ships as source only and compiles against Apache:
#
#   sudo dnf install -y gcc httpd-devel python3.14-devel python3.14-pip
#   sudo python3.14 -m pip install -r requirements_aws.txt
#
# boto3 and botocore must stay on the same version.
